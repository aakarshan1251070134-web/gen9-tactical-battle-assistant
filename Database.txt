"This query demonstrates complex INNER JOIN and LEFT JOIN operations across multiple tables to translate foreign key IDs into human-readable type names and combat stats."

SELECT c.Name, t1.Type_Name AS PrimaryType, IFNULL(t2.Type_Name, 'None') AS SecondaryType, c.Base_Attack
    -> FROM Creatures c
    -> JOIN Types t1 ON c.Primary_Type_ID = t1.Type_ID
    -> LEFT JOIN Types t2 ON c.Secondary_Type_ID = t2.Type_ID
    -> LIMIT 5;
